\section*{A.1 Tenant Context Propagation Across Asynchronous Boundaries}

\begin{lstlisting}[caption={Asynchronous template that propagates the tenant ThreadLocal.}]
public class AsyncTemplate {
  private final Executor executor;

  public CompletableFuture<Void> runAsync(Runnable work) {
    final String tenantId = TenantContext.get();
    return CompletableFuture.runAsync(() -> {
      try {
        TenantContext.set(tenantId);
        work.run();
      } finally {
        TenantContext.clear();
      }
    }, executor);
  }
}
\end{lstlisting}

\section*{A.3 Message Body Micro-Variation}

\begin{lstlisting}[caption={Micro-variation applied to every outbound WhatsApp body.}]
function applyMicroVariations(text: string): string {
  if (Math.random() < 0.3) {
    text = text.endsWith('.') ? text.slice(0, -1) : text + '.';
  }
  if (Math.random() < 0.2) {
    const pos = randomInt(1, text.length - 1);
    text = text.slice(0, pos) + ZWSP + text.slice(pos); // U+200B
  }
  return text;
}
\end{lstlisting}
